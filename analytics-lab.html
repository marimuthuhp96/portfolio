groundwater_analysis.py
import pandas as pd from sklearn.cluster import KMeans from sklearn.preprocessing import StandardScaler # Load groundwater dataset df = pd.read_csv('groundwater_data.csv') # Feature engineering features = ['depth', 'quality_score', 'rainfall'] X = StandardScaler().fit_transform(df[features]) # K-Means clustering (k=3 optimal) kmeans = KMeans(n_clusters=3, random_state=42) df['cluster'] = kmeans.fit_predict(X) # Risk assessment risk_map = {0: 'Critical', 1: 'Moderate', 2: 'Stable'} df['risk_level'] = df['cluster'].map(risk_map) print(df['risk_level'].value_counts()) # Critical: 47 | Moderate: 283 | Stable: 270
sales_cohort_analysis.sql
-- Customer Cohort Retention Analysis WITH cohort AS ( SELECT customer_id, DATE_TRUNC('month', first_order) AS cohort_month, DATE_TRUNC('month', order_date) AS order_month FROM orders o JOIN customers c USING (customer_id) ), retention AS ( SELECT cohort_month, DATEDIFF('month',cohort_month,order_month) AS period, COUNT(DISTINCT customer_id) AS customers FROM cohort GROUP BY 1, 2 ) SELECT cohort_month, ROUND(customers * 100.0 / FIRST_VALUE(customers) OVER (PARTITION BY cohort_month ORDER BY period), 1) AS retention_rate FROM retention;
